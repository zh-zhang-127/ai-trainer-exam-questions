<script>
const ALLQ    = QDATA_PH;
const ALLCATS = CATS_PH;
const ALLSRCS = SRCS_PH;
const ALLKWDS = KWDS_PH;

// ── type helpers ──────────────────────────────────────────────────
const TYPE_LABEL = {single:'单选', multi:'多选', tf:'判断'};
const TYPE_CLS   = {single:'tag-single', multi:'tag-multi', tf:'tag-tf'};

function getCorrect(q) {
  // ans is always a string: "A", "AB", "正确", "错误"
  if (!q.ans) return [];
  if (q.type === 'tf') return [q.ans];
  return q.ans.split(''); // ["A"] or ["A","B"]
}

// ── localStorage ─────────────────────────────────────────────────
const LS_KEY = 'aiquiz2';
let prog = {};
try { prog = JSON.parse(localStorage.getItem(LS_KEY)) || {}; } catch(e) {}

// ── state ─────────────────────────────────────────────────────────
let curCat  = ALLCATS[0] || '';
let curSrc  = '全部';
let curMode = 0;
let queue   = [];
let qi      = 0;
let selOpts = new Set();
let submitted = false;

// ── source filter ─────────────────────────────────────────────────
function buildSrcFilter() {
  const c = document.getElementById('srcFilter');
  c.innerHTML = ['全部', ...ALLSRCS].map(s =>
    `<button class="sfbtn${s===curSrc?' on':''}" onclick="setSrc('${s.replace(/'/g,"\\'")}'">${s}</button>`
  ).join('');
}
function setSrc(s) { curSrc = s; buildSrcFilter(); buildCatList(); buildQueue(); }

// ── filtered pool ─────────────────────────────────────────────────
function filteredQuestions() {
  return ALLQ.filter(q => curSrc === '全部' || q.src === curSrc);
}

// ── category list ─────────────────────────────────────────────────
function buildCatList() {
  const fq = filteredQuestions();
  const cc = {};
  fq.forEach(q => cc[q.cat] = (cc[q.cat]||0)+1);
  const cats = ALLCATS.filter(c => cc[c] > 0);
  if (cc['综合与其他']) cats.push('综合与其他');
  if (!cats.includes(curCat)) curCat = cats[0] || '';
  const el = document.getElementById('catlist');
  el.innerHTML = cats.map(c => {
    const total = cc[c]||0;
    const done  = fq.filter(q => q.cat===c && prog[q.id]!=null).length;
    const pct   = total ? Math.round(done/total*100) : 0;
    return `<div class="ci${c===curCat?' on':''}" onclick="pickCat('${c.replace(/'/g,"\\'")}')">
      ${c}<span class="ci-cnt">${done}/${total}</span>
      <div style="height:3px;background:#e5e7eb;border-radius:2px;margin-top:4px">
        <div style="height:100%;width:${pct}%;background:#7c3aed;border-radius:2px"></div>
      </div></div>`;
  }).join('');
}
function pickCat(c) { curCat = c; buildCatList(); buildQueue(); }

// ── queue ─────────────────────────────────────────────────────────
function buildQueue() {
  const pool = filteredQuestions().filter(q => q.cat === curCat);
  if (curMode === 0)      queue = [...pool];
  else if (curMode === 1) queue = [...pool].sort(() => Math.random()-.5);
  else                    queue = pool.filter(q => prog[q.id] && !prog[q.id].ok);
  qi = 0;
  buildDots();
  showQ();
}
function setMode(m) {
  curMode = m;
  [0,1,2].forEach(i => document.getElementById('m'+i).classList.toggle('on', i===m));
  buildQueue();
}

// ── dots ─────────────────────────────────────────────────────────
function buildDots() {
  document.getElementById('dotbox').innerHTML = queue.map((q,i) => {
    const p = prog[q.id];
    return `<div class="dot${p?(p.ok?' ans-ok':' ans-ng'):''}" id="d${i}" title="第${i+1}题" onclick="jumpTo(${i})"></div>`;
  }).join('');
  highlightDot();
}
function highlightDot() {
  document.querySelectorAll('.dot').forEach(d => d.classList.remove('cur'));
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.add('cur'); d.scrollIntoView({block:'nearest'}); }
}
function jumpTo(i) { qi=i; selOpts.clear(); submitted=false; highlightDot(); showQ(); }

// ── quiz display ──────────────────────────────────────────────────
function showQ() {
  const empty = document.getElementById('empty');
  const sub   = document.getElementById('submit');
  const res   = document.getElementById('result');
  if (!queue.length) {
    empty.style.display='block'; sub.style.display='none'; res.style.display='none';
    document.getElementById('qnum').innerHTML='';
    document.getElementById('qtext').textContent='';
    document.getElementById('opts').innerHTML='';
    document.getElementById('multi-hint').style.display='none';
    return;
  }
  empty.style.display='none'; sub.style.display='inline-block';
  const q = queue[qi];
  const p = prog[q.id];
  submitted = !!p;
  selOpts = new Set(p ? p.ans : []);
  const correct = getCorrect(q);

  document.getElementById('qnum').innerHTML =
    `<span>第 ${qi+1} / ${queue.length} 题</span>
     <span class="qtag ${TYPE_CLS[q.type]||'tag-single'}">${TYPE_LABEL[q.type]||q.type}</span>
     <span class="qtag ${q.src==='泰迪云课堂'?'tag-src-td':'tag-src-ai'}">${q.src}</span>
     <span style="color:#9ca3af;font-size:11px">${q.cat}</span>`;
  document.getElementById('qtext').textContent = q.text;
  document.getElementById('multi-hint').style.display = q.type==='multi'?'block':'none';

  const keys = Object.keys(q.opts);
  document.getElementById('opts').innerHTML = keys.map(k => {
    let cls = 'opt';
    if (submitted) {
      if (correct.includes(k)) cls+=' correct';
      else if (selOpts.has(k)) cls+=' wrong';
    } else if (selOpts.has(k)) cls+=' sel';
    return `<div class="${cls}" id="o${k}" onclick="pick('${k}')">
      <span class="opt-key">${k}</span>
      <span class="opt-txt">${q.opts[k]}</span></div>`;
  }).join('');

  sub.disabled = submitted || selOpts.size===0;
  if (submitted) showResult(q, p.ok); else res.style.display='none';
  updateHeader();
}

function pick(k) {
  if (submitted) return;
  const q = queue[qi];
  if (q.type === 'multi') { selOpts.has(k)?selOpts.delete(k):selOpts.add(k); }
  else { selOpts.clear(); selOpts.add(k); }
  document.querySelectorAll('.opt').forEach(el => {
    el.classList.toggle('sel', selOpts.has(el.id.replace('o','')));
  });
  document.getElementById('submit').disabled = selOpts.size===0;
}

function doSubmit() {
  if (!queue.length || submitted) return;
  const q = queue[qi];
  const correct = getCorrect(q);
  const ok = correct.sort().join('') === [...selOpts].sort().join('');
  prog[q.id] = { ans:[...selOpts], ok };
  try { localStorage.setItem(LS_KEY, JSON.stringify(prog)); } catch(e) {}
  submitted = true;
  document.getElementById('submit').disabled = true;
  Object.keys(q.opts).forEach(k => {
    const el = document.getElementById('o'+k);
    if (!el) return;
    el.classList.remove('sel');
    if (correct.includes(k)) el.classList.add('correct');
    else if (selOpts.has(k)) el.classList.add('wrong');
  });
  showResult(q, ok);
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.remove('ans-ok','ans-ng'); d.classList.add(ok?'ans-ok':'ans-ng'); }
  updateHeader(); buildCatList();
}

function showResult(q, ok) {
  const res = document.getElementById('result');
  res.style.display='block'; res.className = ok?'ok':'ng';
  const correct = getCorrect(q);
  document.getElementById('rh').textContent = ok ? '✅ 回答正确！' : `❌ 回答错误，正确答案：${correct.join('、')}`;
  document.getElementById('rnote').textContent = q.note ? `解析：${q.note}` : '（暂无解析）';
}

function next() {
  if (qi < queue.length-1) { qi++; selOpts.clear(); submitted=false; highlightDot(); showQ(); }
}

// ── header stats ──────────────────────────────────────────────────
function updateHeader() {
  const all  = filteredQuestions();
  const done = all.filter(q => prog[q.id]!=null);
  const ok   = done.filter(q => prog[q.id].ok);
  document.getElementById('hdone').textContent = done.length;
  document.getElementById('htotal').textContent = all.length;
  const acc = done.length ? Math.round(ok.length/done.length*100) : 0;
  document.getElementById('hacc').textContent = done.length ? acc+'%' : '-';
  document.getElementById('hbarf').style.width = (done.length/Math.max(all.length,1)*100)+'%';
}

function resetAll() {
  if (!confirm('确定重置所有进度？')) return;
  prog = {};
  try { localStorage.removeItem(LS_KEY); } catch(e) {}
  buildCatList(); buildQueue(); updateHeader();
}

// ═══════════════════════════════════════════════════════════════════
// ── Knowledge Preview ─────────────────────────────────────────────
// ═══════════════════════════════════════════════════════════════════
// Pre-build keyword regex for highlighting
const KW_PATTERN = ALLKWDS.length
  ? new RegExp('(' + ALLKWDS.map(k => k.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')).join('|') + ')', 'g')
  : null;

function hlText(parent, text) {
  if (!text) return;
  if (!KW_PATTERN) { parent.appendChild(document.createTextNode(text)); return; }
  KW_PATTERN.lastIndex = 0;
  const parts = text.split(KW_PATTERN);
  const kwSet = new Set(ALLKWDS);
  parts.forEach(p => {
    if (!p) return;
    if (kwSet.has(p)) {
      const m = document.createElement('mark'); m.className='kw'; m.textContent=p;
      parent.appendChild(m);
    } else {
      parent.appendChild(document.createTextNode(p));
    }
  });
}

let pvPool = [];
let pvShown = 0;
const PV_BATCH = 60;

function openPreview() {
  // Build selects
  const srcSel = document.getElementById('pvSrcFilter');
  srcSel.innerHTML = '<option value="">全部来源</option>' +
    ALLSRCS.map(s => `<option value="${s}">${s}</option>`).join('');
  const catSel = document.getElementById('pvCatFilter');
  catSel.innerHTML = '<option value="">全部分类</option>' +
    ALLCATS.map(c => `<option value="${c}">${c}</option>`).join('');
  // Pre-set to current context
  catSel.value = curCat || '';
  srcSel.value = (curSrc !== '全部') ? curSrc : '';
  document.getElementById('pvSearch').value = '';
  document.getElementById('pvTypeFilter').value = '';
  document.getElementById('pvOverlay').classList.add('open');
  pvFilter();
}

function closePreview() {
  document.getElementById('pvOverlay').classList.remove('open');
}

function pvBgClick(e) {
  if (e.target.id === 'pvOverlay') closePreview();
}

function pvFilter() {
  const kw   = document.getElementById('pvSearch').value.trim().toLowerCase();
  const type = document.getElementById('pvTypeFilter').value;
  const src  = document.getElementById('pvSrcFilter').value;
  const cat  = document.getElementById('pvCatFilter').value;

  pvPool = ALLQ.filter(q => {
    if (type && q.type !== type) return false;
    if (src  && q.src  !== src)  return false;
    if (cat  && q.cat  !== cat)  return false;
    if (kw) {
      const hay = (q.text+' '+(q.kp||'')+' '+(q.note||'')).toLowerCase();
      if (!hay.includes(kw)) return false;
    }
    return true;
  });

  pvShown = 0;
  // Clear list
  const list = document.getElementById('pvList');
  const empty = document.getElementById('pvEmpty');
  [...list.children].forEach(c => { if (c.id !== 'pvEmpty') c.remove(); });
  document.getElementById('pvMore') && document.getElementById('pvMore').remove();

  const withNote = pvPool.filter(q => q.note).length;
  document.getElementById('pvStats').innerHTML =
    `共 <b>${pvPool.length}</b> 条知识点 &nbsp;·&nbsp; 有解析 <b>${withNote}</b> 条`;

  if (!pvPool.length) { empty.style.display='block'; return; }
  empty.style.display = 'none';
  pvLoadMore();
}

function pvLoadMore() {
  const list  = document.getElementById('pvList');
  const empty = document.getElementById('pvEmpty');
  const end   = Math.min(pvShown + PV_BATCH, pvPool.length);
  const frag  = document.createDocumentFragment();

  for (let i = pvShown; i < end; i++) {
    frag.appendChild(makeKPCard(pvPool[i], i));
  }
  pvShown = end;

  // Remove old "more" button
  const old = document.getElementById('pvMore');
  if (old) old.remove();

  list.insertBefore(frag, empty);

  if (pvShown < pvPool.length) {
    const btn = document.createElement('button');
    btn.id = 'pvMore';
    btn.textContent = `显示更多（还有 ${pvPool.length - pvShown} 条）`;
    btn.onclick = pvLoadMore;
    list.insertBefore(btn, empty);
  }
}

function makeKPCard(q, idx) {
  const wrap = document.createElement('div');
  wrap.className = 'kpc';

  // Index number
  const idxEl = document.createElement('div');
  idxEl.className = 'kpc-idx';
  idxEl.textContent = idx + 1;
  wrap.appendChild(idxEl);

  const body = document.createElement('div');
  body.className = 'kpc-body';

  // Type + source meta tags
  const meta = document.createElement('div');
  meta.className = 'kpc-meta';
  const t1 = document.createElement('span');
  t1.className = `qtag ${TYPE_CLS[q.type]||'tag-single'}`;
  t1.textContent = TYPE_LABEL[q.type] || q.type;
  const t2 = document.createElement('span');
  t2.className = `qtag ${q.src==='泰迪云课堂'?'tag-src-td':'tag-src-ai'}`;
  t2.textContent = q.src;
  meta.append(t1, t2);
  body.appendChild(meta);

  // One-sentence knowledge point
  const kpEl = document.createElement('div');
  kpEl.className = 'kpc-kp';
  hlText(kpEl, q.kp || q.text);
  body.appendChild(kpEl);

  // Category (small, below)
  const catEl = document.createElement('div');
  catEl.className = 'kpc-cat';
  catEl.textContent = q.cat;
  body.appendChild(catEl);

  wrap.appendChild(body);
  return wrap;
}
