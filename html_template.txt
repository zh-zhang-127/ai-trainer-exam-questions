<script>
const ALLQ    = JSON.parse(document.getElementById('_d0').textContent);
const ALLCATS = JSON.parse(document.getElementById('_d1').textContent);
const ALLSRCS = JSON.parse(document.getElementById('_d2').textContent);
const ALLKWDS = JSON.parse(document.getElementById('_d3').textContent);

// ── type helpers ──────────────────────────────────────────────────
const TYPE_LABEL = {single:'单选', multi:'多选', tf:'判断'};
const TYPE_CLS   = {single:'tag-single', multi:'tag-multi', tf:'tag-tf'};

function getCorrect(q) {
  // ans is always a string: "A", "AB", "正确", "错误"
  if (!q.ans) return [];
  if (q.type === 'tf') return [q.ans];
  return q.ans.split(''); // ["A"] or ["A","B"]
}

// ── localStorage ─────────────────────────────────────────────────
const LS_KEY = 'aiquiz2';
let prog = {};
try { prog = JSON.parse(localStorage.getItem(LS_KEY)) || {}; } catch(e) {}

// ── state ─────────────────────────────────────────────────────────
const CHUNK = 20;
let curCat  = ALLCATS[0] || '';
let curSrc  = '全部';
let curMode = 0;
let queue   = [];
let wStart  = 0;
let wEnd    = 0;
let pageSelections = {};

// ── source filter ─────────────────────────────────────────────────
function buildSrcFilter() {
  const c = document.getElementById('srcFilter');
  c.innerHTML = ['全部', ...ALLSRCS].map(s =>
    '<button class="sfbtn' + (s===curSrc?' on':'') + '" onclick="setSrc(\'' + s.replace(/'/g,"\\'") + '\')">' + s + '<\/button>'
  ).join('');
}
function setSrc(s) { curSrc = s; buildSrcFilter(); buildCatList(); buildQueue(); }

// ── filtered pool ─────────────────────────────────────────────────
function filteredQuestions() {
  return ALLQ.filter(q => curSrc === '全部' || q.src === curSrc);
}

// ── category list ─────────────────────────────────────────────────
function buildCatList() {
  const fq = filteredQuestions();
  const cc = {};
  fq.forEach(q => cc[q.cat] = (cc[q.cat]||0)+1);
  const cats = ALLCATS.filter(c => cc[c] > 0);
  if (cc['综合与其他']) cats.push('综合与其他');
  if (!cats.includes(curCat)) curCat = cats[0] || '';
  const el = document.getElementById('catlist');
  el.innerHTML = cats.map(c => {
    const total = cc[c]||0;
    const done  = fq.filter(q => q.cat===c && prog[q.id]!=null).length;
    const pct   = total ? Math.round(done/total*100) : 0;
    return '<div class="ci' + (c===curCat?' on':'') + '" onclick="pickCat(\'' + c.replace(/'/g,"\\'") + '\')">\n' +
      '  ' + c + '<span class="ci-cnt">' + done + '\/' + total + '<\/span>\n' +
      '  <div style="height:3px;background:#e5e7eb;border-radius:2px;margin-top:4px">\n' +
      '    <div style="height:100%;width:' + pct + '%;background:#7c3aed;border-radius:2px"><\/div>\n' +
      '  <\/div><\/div>';
  }).join('');
}
function pickCat(c) { curCat = c; buildCatList(); buildQueue(); }

// ── queue ─────────────────────────────────────────────────────────
function buildQueue() {
  const pool = filteredQuestions().filter(q => q.cat === curCat);
  if (curMode === 0)      queue = [...pool];
  else if (curMode === 1) queue = [...pool].sort(() => Math.random()-.5);
  else                    queue = pool.filter(q => prog[q.id] && !prog[q.id].ok);
  pageSelections = {};
  buildDots();
  initWindow();
}
function setMode(m) {
  curMode = m;
  [0,1,2].forEach(i => document.getElementById('m'+i).classList.toggle('on', i===m));
  buildQueue();
}

// ── dots ─────────────────────────────────────────────────────────
function buildDots() {
  document.getElementById('dotbox').innerHTML = queue.map(function(q, i) {
    var p = prog[q.id];
    return '<div class="dot' + (p?(p.ok?' ans-ok':' ans-ng'):'') + '" id="d' + i + '" title="第' + (i+1) + '题" onclick="jumpTo(' + i + ')">' + (i+1) + '<\/div>';
  }).join('');
  highlightWindowDots();
}
function highlightWindowDots() {
  var mid = document.getElementById('d' + Math.floor((wStart + wEnd) / 2));
  if (mid) mid.scrollIntoView({block:'nearest'});
}
function jumpTo(i) {
  var dot = document.getElementById('d' + i);
  if (dot) {
    dot.classList.remove('clicked');
    void dot.offsetWidth;
    dot.classList.add('clicked');
    dot.addEventListener('animationend', function() { dot.classList.remove('clicked'); }, {once:true});
  }
  if (i >= wStart && i < wEnd) {
    var el = document.getElementById('qc' + i);
    if (el) el.scrollIntoView({behavior:'smooth', block:'start'});
  } else {
    initWindow(i);
    setTimeout(function() {
      var el = document.getElementById('qc' + i);
      if (el) el.scrollIntoView({behavior:'smooth', block:'start'});
    }, 30);
  }
}

// ── window init & infinite scroll ────────────────────────────────
function initWindow(centerIdx) {
  var listEl = document.getElementById('qlist');
  listEl.innerHTML = '';
  if (!queue.length) {
    var empty = document.createElement('div');
    empty.style.cssText = 'text-align:center;color:#9ca3af;font-size:15px;padding:40px';
    empty.textContent = '该分类暂无题目';
    listEl.appendChild(empty);
    wStart = 0; wEnd = 0;
    updateInfo(); updateHeader();
    return;
  }
  var c = centerIdx != null ? centerIdx : 0;
  wStart = Math.max(0, c - CHUNK);
  wEnd   = Math.min(queue.length, c + CHUNK);
  for (var i = wStart; i < wEnd; i++) listEl.appendChild(buildQCard(i));
  // scroll so centerIdx is visible; default scroll to top
  if (centerIdx != null) {
    var el = document.getElementById('qc' + centerIdx);
    if (el) el.scrollIntoView({block:'center'});
  } else {
    listEl.scrollTop = 0;
  }
  updateInfo(); updateHeader(); highlightWindowDots();
}

function appendChunk() {
  if (wEnd >= queue.length) return;
  var listEl = document.getElementById('qlist');
  var newEnd = Math.min(wEnd + CHUNK, queue.length);
  for (var i = wEnd; i < newEnd; i++) listEl.appendChild(buildQCard(i));
  wEnd = newEnd;
  // trim front if window > 3 chunks
  if (wEnd - wStart > CHUNK * 3) {
    var oldScrollTop = listEl.scrollTop;
    var oldHeight = listEl.scrollHeight;
    for (var j = 0; j < CHUNK && listEl.firstChild; j++) listEl.removeChild(listEl.firstChild);
    wStart += CHUNK;
    listEl.scrollTop = oldScrollTop - (oldHeight - listEl.scrollHeight);
  }
  updateInfo(); highlightWindowDots();
}

function prependChunk() {
  if (wStart <= 0) return;
  var listEl = document.getElementById('qlist');
  var newStart = Math.max(0, wStart - CHUNK);
  var oldScrollTop = listEl.scrollTop;
  var oldHeight = listEl.scrollHeight;
  var frag = document.createDocumentFragment();
  for (var i = newStart; i < wStart; i++) frag.appendChild(buildQCard(i));
  listEl.insertBefore(frag, listEl.firstChild);
  wStart = newStart;
  listEl.scrollTop = oldScrollTop + (listEl.scrollHeight - oldHeight);
  // trim back if window > 3 chunks
  if (wEnd - wStart > CHUNK * 3) {
    for (var j = 0; j < CHUNK && listEl.lastChild; j++) listEl.removeChild(listEl.lastChild);
    wEnd -= CHUNK;
  }
  updateInfo(); highlightWindowDots();
}

// ── info bar ──────────────────────────────────────────────────────
function countUnsub() {
  var n = 0;
  for (var i = wStart; i < wEnd; i++) {
    if (!prog[queue[i].id] && pageSelections[i] && pageSelections[i].size > 0) n++;
  }
  return n;
}
function updateInfo() {
  var el = document.getElementById('qinfo');
  if (!el || !queue.length) { if (el) el.innerHTML = ''; return; }
  var unsub = countUnsub();
  el.innerHTML = '<span class="pnav-info">共 ' + queue.length + ' 题<\/span>' +
    '<button class="btn-sub-page"' + (unsub===0?' disabled':'') + ' onclick="submitVisible()">提交已选（' + unsub + ' 题）<\/button>';
}

function submitVisible() {
  for (var i = wStart; i < wEnd; i++) {
    if (!prog[queue[i].id] && pageSelections[i] && pageSelections[i].size > 0) submitOne(i);
  }
}

// ── scroll infinite load ──────────────────────────────────────────
(function() {
  document.addEventListener('DOMContentLoaded', function() {
    var listEl = document.getElementById('qlist');
    if (!listEl) return;
    var lastST = 0;
    listEl.addEventListener('scroll', function() {
      var st = listEl.scrollTop;
      var threshold = 120;
      if (st > lastST && st + listEl.clientHeight >= listEl.scrollHeight - threshold) appendChunk();
      else if (st < lastST && st <= threshold) prependChunk();
      lastST = st;
    });
  });
})();

function buildQCard(i) {
  var q = queue[i];
  var p = prog[q.id];
  var done = !!p;
  var sel = pageSelections[i] || new Set();
  var correct = getCorrect(q);

  var card = document.createElement('div');
  card.className = 'qcard' + (done ? (p.ok ? ' done-ok' : ' done-ng') : '');
  card.id = 'qc' + i;

  // Header
  var hd = document.createElement('div');
  hd.className = 'qcard-hd';
  var numSp = document.createElement('span');
  numSp.className = 'qcard-num';
  numSp.textContent = '第 ' + (i+1) + ' 题';
  var typeSp = document.createElement('span');
  typeSp.className = 'qtag ' + (TYPE_CLS[q.type]||'tag-single');
  typeSp.textContent = TYPE_LABEL[q.type] || q.type;
  var srcSp = document.createElement('span');
  srcSp.className = 'qtag ' + (q.src==='泰迪云课堂' ? 'tag-src-td' : 'tag-src-ai');
  srcSp.textContent = q.src;
  var statusSp = document.createElement('span');
  statusSp.className = 'qcard-status ' + (done ? (p.ok ? 'qs-ok' : 'qs-ng') : (sel.size > 0 ? 'qs-sel' : 'qs-pending'));
  statusSp.id = 'qs' + i;
  statusSp.textContent = done ? (p.ok ? '✅ 正确' : '❌ 错误') : (sel.size > 0 ? '已选择' : '未作答');
  hd.append(numSp, typeSp, srcSp, statusSp);
  card.appendChild(hd);

  // Multi hint
  if (q.type === 'multi') {
    var hint = document.createElement('div');
    hint.className = 'multi-hint';
    hint.textContent = '（多选题，请选择所有正确答案）';
    card.appendChild(hint);
  }

  // Question text
  var txt = document.createElement('div');
  txt.className = 'qcard-text';
  txt.textContent = q.text;
  card.appendChild(txt);

  // Options
  var optsEl = document.createElement('div');
  optsEl.className = 'opts';
  optsEl.id = 'opts' + i;
  var keys = Object.keys(q.opts);
  keys.forEach(function(k) {
    var optEl = document.createElement('div');
    var cls = 'opt';
    if (done) {
      if (correct.indexOf(k) >= 0) cls += ' correct';
      else if (p.ans.indexOf(k) >= 0) cls += ' wrong';
    } else if (sel.has(k)) {
      cls += ' sel';
    }
    optEl.className = cls;
    optEl.id = 'o' + i + '_' + k;
    if (!done) {
      (function(key) {
        optEl.onclick = function() { pickOpt(i, key); };
      })(k);
    }
    var keySp = document.createElement('span');
    keySp.className = 'opt-key';
    keySp.textContent = k;
    var valSp = document.createElement('span');
    valSp.className = 'opt-txt';
    valSp.textContent = q.opts[k];
    optEl.append(keySp, valSp);
    optsEl.appendChild(optEl);
  });
  card.appendChild(optsEl);

  // Submit button (only if not yet done)
  if (!done) {
    var btns = document.createElement('div');
    btns.className = 'qcard-btns';
    var subBtn = document.createElement('button');
    subBtn.className = 'qcard-sub';
    subBtn.id = 'sb' + i;
    subBtn.textContent = '立即提交';
    subBtn.disabled = sel.size === 0;
    (function(idx) { subBtn.onclick = function() { submitOne(idx); }; })(i);
    btns.appendChild(subBtn);
    card.appendChild(btns);
  }

  // Result (if already submitted)
  if (done) {
    card.appendChild(buildResultEl(q, p.ok));
  }

  return card;
}

function buildResultEl(q, ok) {
  var res = document.createElement('div');
  res.className = 'qcard-result ' + (ok ? 'ok' : 'ng');
  var rh = document.createElement('span');
  rh.className = 'rh';
  var correct = getCorrect(q);
  rh.textContent = ok ? '✅ 回答正确！' : '❌ 回答错误，正确答案：' + correct.join('、');
  res.appendChild(rh);
  if (q.note) {
    var note = document.createElement('div');
    note.className = 'note';
    note.textContent = '解析：' + q.note;
    res.appendChild(note);
  }
  return res;
}

// ── interaction ───────────────────────────────────────────────────
function pickOpt(i, key) {
  var q = queue[i];
  if (!pageSelections[i]) pageSelections[i] = new Set();
  var sel = pageSelections[i];
  if (q.type === 'multi') {
    if (sel.has(key)) sel.delete(key); else sel.add(key);
  } else {
    sel.clear(); sel.add(key);
  }
  Object.keys(q.opts).forEach(function(k) {
    var el = document.getElementById('o' + i + '_' + k);
    if (el) el.className = 'opt' + (sel.has(k) ? ' sel' : '');
  });
  var sb = document.getElementById('sb' + i);
  if (sb) sb.disabled = sel.size === 0;
  var qs = document.getElementById('qs' + i);
  if (qs) {
    qs.className = 'qcard-status ' + (sel.size > 0 ? 'qs-sel' : 'qs-pending');
    qs.textContent = sel.size > 0 ? '已选择' : '未作答';
  }
  var dot = document.getElementById('d' + i);
  if (dot) {
    dot.classList.remove('ans-sel');
    if (sel.size > 0) dot.classList.add('ans-sel');
  }
  updateInfo();
}

function submitOne(i) {
  var q = queue[i];
  var sel = pageSelections[i];
  if (!sel || sel.size === 0) return;
  var correct = getCorrect(q);
  var selArr = [...sel].sort();
  var ok = correct.slice().sort().join('') === selArr.join('');
  prog[q.id] = { ans: selArr, ok: ok };
  try { localStorage.setItem(LS_KEY, JSON.stringify(prog)); } catch(e) {}

  var card = document.getElementById('qc' + i);
  if (card) {
    card.className = 'qcard ' + (ok ? 'done-ok' : 'done-ng');
    var btns = card.querySelector('.qcard-btns');
    if (btns) btns.remove();
    Object.keys(q.opts).forEach(function(k) {
      var el = document.getElementById('o' + i + '_' + k);
      if (!el) return;
      el.onclick = null;
      el.className = 'opt';
      if (correct.indexOf(k) >= 0) el.classList.add('correct');
      else if (sel.has(k)) el.classList.add('wrong');
    });
    card.appendChild(buildResultEl(q, ok));
  }

  var qs = document.getElementById('qs' + i);
  if (qs) {
    qs.className = 'qcard-status ' + (ok ? 'qs-ok' : 'qs-ng');
    qs.textContent = ok ? '✅ 正确' : '❌ 错误';
  }
  var dot = document.getElementById('d' + i);
  if (dot) { dot.classList.remove('ans-sel','ans-ok','ans-ng'); dot.classList.add(ok ? 'ans-ok' : 'ans-ng'); }
  updateInfo();
  updateHeader();
  buildCatList();
}

function submitVisible() {
  for (var i = wStart; i < wEnd; i++) {
    if (!prog[queue[i].id] && pageSelections[i] && pageSelections[i].size > 0) submitOne(i);
  }
}

// ── header stats ──────────────────────────────────────────────────
function updateHeader() {
  const all  = filteredQuestions();
  const done = all.filter(q => prog[q.id]!=null);
  const ok   = done.filter(q => prog[q.id].ok);
  document.getElementById('hdone').textContent = done.length;
  document.getElementById('htotal').textContent = all.length;
  const acc = done.length ? Math.round(ok.length/done.length*100) : 0;
  document.getElementById('hacc').textContent = done.length ? acc+'%' : '-';
  document.getElementById('hbarf').style.width = (done.length/Math.max(all.length,1)*100)+'%';
}

function resetAll() {
  if (!confirm('确定重置所有进度？')) return;
  prog = {};
  pageSelections = {};
  try { localStorage.removeItem(LS_KEY); } catch(e) {}
  buildDots(); buildCatList(); initWindow(); updateHeader();
}

// Pre-build keyword regex for highlighting
const KW_PATTERN = ALLKWDS.length
  ? new RegExp('(' + ALLKWDS.map(k => k.replace(/[.*+?^${}()|[\]\\]/g,'\\$&')).join('|') + ')', 'g')
  : null;

function hlText(parent, text) {
  if (!text) return;
  if (!KW_PATTERN) { parent.appendChild(document.createTextNode(text)); return; }
  KW_PATTERN.lastIndex = 0;
  const parts = text.split(KW_PATTERN);
  const kwSet = new Set(ALLKWDS);
  parts.forEach(p => {
    if (!p) return;
    if (kwSet.has(p)) {
      const m = document.createElement('mark'); m.className='kw'; m.textContent=p;
      parent.appendChild(m);
    } else {
      parent.appendChild(document.createTextNode(p));
    }
  });
}

function pvFilter() {
  const kw   = document.getElementById('pvSearch').value.trim().toLowerCase();
  const type = document.getElementById('pvTypeFilter').value;
  const src  = document.getElementById('pvSrcFilter').value;
  const cat  = document.getElementById('pvCatFilter').value;

  pvPool = ALLQ.filter(q => {
    if (type && q.type !== type) return false;
    if (src  && q.src  !== src)  return false;
    if (cat  && q.cat  !== cat)  return false;
    if (kw) {
      const hay = (q.text+' '+(q.kp||'')+' '+(q.note||'')).toLowerCase();
      if (!hay.includes(kw)) return false;
    }
    return true;
  });

  pvShown = 0;
  // Clear list
  const list = document.getElementById('pvList');
  const empty = document.getElementById('pvEmpty');
  [...list.children].forEach(c => { if (c.id !== 'pvEmpty') c.remove(); });
  document.getElementById('pvMore') && document.getElementById('pvMore').remove();

  const withNote = pvPool.filter(q => q.note).length;
  document.getElementById('pvStats').innerHTML =
    '共 <b>' + pvPool.length + '<\/b> 条知识点 &nbsp;·&nbsp; 有解析 <b>' + withNote + '<\/b> 条';

  if (!pvPool.length) { empty.style.display='block'; return; }
  empty.style.display = 'none';
  pvLoadMore();
}

function pvLoadMore() {
  const list  = document.getElementById('pvList');
  const empty = document.getElementById('pvEmpty');
  const end   = Math.min(pvShown + PV_BATCH, pvPool.length);
  const frag  = document.createDocumentFragment();

  for (let i = pvShown; i < end; i++) {
    frag.appendChild(makeKPCard(pvPool[i], i));
  }
  pvShown = end;

  // Remove old "more" button
  const old = document.getElementById('pvMore');
  if (old) old.remove();

  list.insertBefore(frag, empty);

  if (pvShown < pvPool.length) {
    const btn = document.createElement('button');
    btn.id = 'pvMore';
    btn.textContent = '显示更多（还有 ' + (pvPool.length - pvShown) + ' 条）';
    btn.onclick = pvLoadMore;
    list.insertBefore(btn, empty);
  }
}

function makeKPCard(q, idx) {
  const wrap = document.createElement('div');
  wrap.className = 'kpc';

  // Index number
  const idxEl = document.createElement('div');
  idxEl.className = 'kpc-idx';
  idxEl.textContent = idx + 1;
  wrap.appendChild(idxEl);

  const body = document.createElement('div');
  body.className = 'kpc-body';

  // Type + source meta tags
  const meta = document.createElement('div');
  meta.className = 'kpc-meta';
  const t1 = document.createElement('span');
  t1.className = 'qtag ' + (TYPE_CLS[q.type]||'tag-single');
  t1.textContent = TYPE_LABEL[q.type] || q.type;
  const t2 = document.createElement('span');
  t2.className = 'qtag ' + (q.src==='泰迪云课堂'?'tag-src-td':'tag-src-ai');
  t2.textContent = q.src;
  meta.append(t1, t2);
  body.appendChild(meta);

  // One-sentence knowledge point
  const kpEl = document.createElement('div');
  kpEl.className = 'kpc-kp';
  hlText(kpEl, q.kp || q.text);
  body.appendChild(kpEl);

  // Category (small, below)
  const catEl = document.createElement('div');
  catEl.className = 'kpc-cat';
  catEl.textContent = q.cat;
  body.appendChild(catEl);

  wrap.appendChild(body);
  return wrap;
}
