<script>
const ALLQ = QDATA_PH;
const ALLCATS = CATS_PH;
const ALLSRCS = SRCS_PH;

const LS_KEY = 'aiquiz2';
let prog = {};
try { prog = JSON.parse(localStorage.getItem(LS_KEY)) || {}; } catch(e) {}

let curCat = ALLCATS[0] || '';
let curSrc = '全部';
let curMode = 0; // 0=seq 1=rand 2=wrong
let queue = [];
let qi = 0;
let selOpts = new Set();
let submitted = false;

// ─── source filter UI ───────────────────────────────────────────
function buildSrcFilter() {
  const c = document.getElementById('srcFilter');
  const sources = ['全部', ...ALLSRCS];
  c.innerHTML = sources.map(s => {
    const active = s === curSrc ? ' on' : '';
    return `<button class="sfbtn${active}" onclick="setSrc('${s}')">${s}</button>`;
  }).join('');
}

function setSrc(s) {
  curSrc = s;
  buildSrcFilter();
  buildCatList();
  buildQueue();
}

// ─── category list ───────────────────────────────────────────────
function filteredQuestions() {
  return ALLQ.filter(q => curSrc === '全部' || q.src === curSrc);
}

function buildCatList() {
  const fq = filteredQuestions();
  const cc = {};
  fq.forEach(q => cc[q.cat] = (cc[q.cat]||0)+1);
  const cats = ALLCATS.filter(c => cc[c] > 0);
  if (cc['综合与其他']) cats.push('综合与其他');
  if (!cats.includes(curCat)) curCat = cats[0] || '';

  const el = document.getElementById('catlist');
  el.innerHTML = cats.map(c => {
    const total = cc[c]||0;
    const done  = fq.filter(q => q.cat===c && prog[q.id]!=null).length;
    const pct   = total ? Math.round(done/total*100) : 0;
    const on = c===curCat?' on':'';
    return `<div class="ci${on}" onclick="pickCat('${c}')">
      ${c}
      <span class="ci-cnt">${done}/${total}</span>
      <div style="height:3px;background:#e5e7eb;border-radius:2px;margin-top:4px">
        <div style="height:100%;width:${pct}%;background:#7c3aed;border-radius:2px"></div>
      </div>
    </div>`;
  }).join('');
}

function pickCat(c) {
  curCat = c;
  buildCatList();
  buildQueue();
}

// ─── queue ───────────────────────────────────────────────────────
function buildQueue() {
  const pool = filteredQuestions().filter(q => q.cat === curCat);
  if (curMode === 0) {
    queue = [...pool];
  } else if (curMode === 1) {
    queue = [...pool].sort(() => Math.random()-.5);
  } else {
    queue = pool.filter(q => prog[q.id] && !prog[q.id].ok);
    if (!queue.length) queue = [...pool].filter(q => prog[q.id] && !prog[q.id].ok);
  }
  qi = 0;
  buildDots();
  showQ();
}

// ─── mode buttons ────────────────────────────────────────────────
function setMode(m) {
  curMode = m;
  [0,1,2].forEach(i => document.getElementById('m'+i).classList.toggle('on', i===m));
  buildQueue();
}

// ─── dots ────────────────────────────────────────────────────────
function buildDots() {
  const box = document.getElementById('dotbox');
  box.innerHTML = queue.map((q, i) => {
    const p = prog[q.id];
    const cls = p ? (p.ok ? 'ans-ok' : 'ans-ng') : '';
    return `<div class="dot ${cls}" id="d${i}" title="第${i+1}题" onclick="jumpTo(${i})"></div>`;
  }).join('');
  highlightDot();
}

function highlightDot() {
  document.querySelectorAll('.dot').forEach(d => d.classList.remove('cur'));
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.add('cur'); d.scrollIntoView({block:'nearest'}); }
}

function jumpTo(i) {
  qi = i;
  selOpts.clear();
  submitted = false;
  highlightDot();
  showQ();
}

// ─── question display ────────────────────────────────────────────
function srcTag(src) {
  if (src === '泰迪云课堂') return '<span class="qtag tag-src-td">泰迪云课堂</span>';
  return '<span class="qtag tag-src-ai">AI训练师题库</span>';
}

function showQ() {
  const empty = document.getElementById('empty');
  const qnum  = document.getElementById('qnum');
  const qtext = document.getElementById('qtext');
  const optsEl= document.getElementById('opts');
  const hint  = document.getElementById('multi-hint');
  const sub   = document.getElementById('submit');
  const res   = document.getElementById('result');

  if (!queue.length) {
    empty.style.display = 'block';
    qnum.innerHTML = '';
    qtext.textContent = '';
    optsEl.innerHTML = '';
    hint.style.display = 'none';
    sub.style.display = 'none';
    res.style.display = 'none';
    return;
  }
  empty.style.display = 'none';
  sub.style.display = 'inline-block';

  const q = queue[qi];
  const typeCls = {'单选':'tag-单选','多选':'tag-多选','判断':'tag-判断'}[q.type]||'tag-单选';
  qnum.innerHTML = `<span>第 ${qi+1} / ${queue.length} 题</span>
    <span class="qtag ${typeCls}">${q.type}</span>
    ${srcTag(q.src)}
    <span style="color:#9ca3af;font-size:11px">${q.cat}</span>`;
  qtext.textContent = q.text;
  hint.style.display = q.type==='多选' ? 'block' : 'none';

  const keys = Object.keys(q.opts);
  const p = prog[q.id];
  submitted = !!p;
  selOpts = new Set(p ? p.ans : []);

  optsEl.innerHTML = keys.map(k => {
    let cls = 'opt';
    if (submitted) {
      const correct = Array.isArray(q.ans) ? q.ans : [q.ans];
      if (correct.includes(k)) cls += ' correct';
      else if (selOpts.has(k)) cls += ' wrong';
    } else if (selOpts.has(k)) {
      cls += ' sel';
    }
    return `<div class="${cls}" id="o${k}" onclick="pick('${k}')">
      <span class="opt-key">${k}</span>
      <span class="opt-txt">${q.opts[k]}</span>
    </div>`;
  }).join('');

  sub.disabled = submitted || selOpts.size === 0;
  if (submitted) showResult(q, p.ok);
  else { res.style.display = 'none'; }
  updateHeader();
}

function pick(k) {
  if (submitted) return;
  const q = queue[qi];
  if (q.type === '多选') {
    selOpts.has(k) ? selOpts.delete(k) : selOpts.add(k);
  } else {
    selOpts.clear(); selOpts.add(k);
  }
  document.querySelectorAll('.opt').forEach(el => {
    const key = el.id.replace('o','');
    el.classList.toggle('sel', selOpts.has(key));
  });
  document.getElementById('submit').disabled = selOpts.size === 0;
}

function submit() {
  if (!queue.length || submitted) return;
  const q = queue[qi];
  const correct = Array.isArray(q.ans) ? [...q.ans].sort().join('') : q.ans;
  const given   = [...selOpts].sort().join('');
  const ok = correct === given;
  prog[q.id] = { ans: [...selOpts], ok };
  try { localStorage.setItem(LS_KEY, JSON.stringify(prog)); } catch(e) {}
  submitted = true;
  document.getElementById('submit').disabled = true;
  // color options
  Object.keys(q.opts).forEach(k => {
    const el = document.getElementById('o'+k);
    if (!el) return;
    el.classList.remove('sel');
    const corArr = Array.isArray(q.ans) ? q.ans : [q.ans];
    if (corArr.includes(k)) el.classList.add('correct');
    else if (selOpts.has(k)) el.classList.add('wrong');
  });
  showResult(q, ok);
  // update dot
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.remove('ans-ok','ans-ng'); d.classList.add(ok?'ans-ok':'ans-ng'); }
  updateHeader();
  buildCatList();
}

function showResult(q, ok) {
  const res = document.getElementById('result');
  const rh  = document.getElementById('rh');
  const rn  = document.getElementById('rnote');
  res.style.display = 'block';
  res.className = ok ? 'ok' : 'ng';
  const correctStr = Array.isArray(q.ans) ? q.ans.join('、') : q.ans;
  rh.textContent = ok ? '✅ 回答正确！' : `❌ 回答错误，正确答案：${correctStr}`;
  rn.textContent = q.note ? `解析：${q.note}` : '（暂无解析）';
}

function next() {
  if (qi < queue.length - 1) {
    qi++;
    selOpts.clear();
    submitted = false;
    highlightDot();
    showQ();
  }
}

// ─── header stats ────────────────────────────────────────────────
function updateHeader() {
  const all = filteredQuestions();
  const done = all.filter(q => prog[q.id] != null);
  const ok   = done.filter(q => prog[q.id].ok);
  document.getElementById('hdone').textContent = done.length;
  document.getElementById('htotal').textContent = all.length;
  const acc = done.length ? Math.round(ok.length/done.length*100) : 0;
  document.getElementById('hacc').textContent = done.length ? acc+'%' : '-';
  document.getElementById('hbarf').style.width = (done.length/Math.max(all.length,1)*100)+'%';
}

function resetAll() {
  if (!confirm('确定重置所有进度？')) return;
  prog = {};
  try { localStorage.removeItem(LS_KEY); } catch(e) {}
  buildCatList();
  buildQueue();
  updateHeader();
}

// ─── init ────────────────────────────────────────────────────────
buildSrcFilter();
buildCatList();
buildQueue();
updateHeader();
</script>
