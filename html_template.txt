<script>
const ALLQ = QDATA_PH;
const ALLCATS = CATS_PH;
const ALLSRCS = SRCS_PH;

const LS_KEY = 'aiquiz2';
let prog = {};
try { prog = JSON.parse(localStorage.getItem(LS_KEY)) || {}; } catch(e) {}

let curCat = ALLCATS[0] || '';
let curSrc = '全部';
let curMode = 0; // 0=seq 1=rand 2=wrong
let queue = [];
let qi = 0;
let selOpts = new Set();
let submitted = false;

// ─── source filter UI ───────────────────────────────────────────
function buildSrcFilter() {
  const c = document.getElementById('srcFilter');
  const sources = ['全部', ...ALLSRCS];
  c.innerHTML = sources.map(s => {
    const active = s === curSrc ? ' on' : '';
    return `<button class="sfbtn${active}" onclick="setSrc('${s}')">${s}</button>`;
  }).join('');
}

function setSrc(s) {
  curSrc = s;
  buildSrcFilter();
  buildCatList();
  buildQueue();
}

// ─── category list ───────────────────────────────────────────────
function filteredQuestions() {
  return ALLQ.filter(q => curSrc === '全部' || q.src === curSrc);
}

function buildCatList() {
  const fq = filteredQuestions();
  const cc = {};
  fq.forEach(q => cc[q.cat] = (cc[q.cat]||0)+1);
  const cats = ALLCATS.filter(c => cc[c] > 0);
  if (cc['综合与其他']) cats.push('综合与其他');
  if (!cats.includes(curCat)) curCat = cats[0] || '';

  const el = document.getElementById('catlist');
  el.innerHTML = cats.map(c => {
    const total = cc[c]||0;
    const done  = fq.filter(q => q.cat===c && prog[q.id]!=null).length;
    const pct   = total ? Math.round(done/total*100) : 0;
    const on = c===curCat?' on':'';
    return `<div class="ci${on}" onclick="pickCat('${c}')">
      ${c}
      <span class="ci-cnt">${done}/${total}</span>
      <div style="height:3px;background:#e5e7eb;border-radius:2px;margin-top:4px">
        <div style="height:100%;width:${pct}%;background:#7c3aed;border-radius:2px"></div>
      </div>
    </div>`;
  }).join('');
}

function pickCat(c) {
  curCat = c;
  buildCatList();
  buildQueue();
}

// ─── queue ───────────────────────────────────────────────────────
function buildQueue() {
  const pool = filteredQuestions().filter(q => q.cat === curCat);
  if (curMode === 0) {
    queue = [...pool];
  } else if (curMode === 1) {
    queue = [...pool].sort(() => Math.random()-.5);
  } else {
    queue = pool.filter(q => prog[q.id] && !prog[q.id].ok);
    if (!queue.length) queue = [...pool].filter(q => prog[q.id] && !prog[q.id].ok);
  }
  qi = 0;
  buildDots();
  showQ();
}

// ─── mode buttons ────────────────────────────────────────────────
function setMode(m) {
  curMode = m;
  [0,1,2].forEach(i => document.getElementById('m'+i).classList.toggle('on', i===m));
  buildQueue();
}

// ─── dots ────────────────────────────────────────────────────────
function buildDots() {
  const box = document.getElementById('dotbox');
  box.innerHTML = queue.map((q, i) => {
    const p = prog[q.id];
    const cls = p ? (p.ok ? 'ans-ok' : 'ans-ng') : '';
    return `<div class="dot ${cls}" id="d${i}" title="第${i+1}题" onclick="jumpTo(${i})"></div>`;
  }).join('');
  highlightDot();
}

function highlightDot() {
  document.querySelectorAll('.dot').forEach(d => d.classList.remove('cur'));
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.add('cur'); d.scrollIntoView({block:'nearest'}); }
}

function jumpTo(i) {
  qi = i;
  selOpts.clear();
  submitted = false;
  highlightDot();
  showQ();
}

// ─── question display ────────────────────────────────────────────
function srcTag(src) {
  if (src === '泰迪云课堂') return '<span class="qtag tag-src-td">泰迪云课堂</span>';
  return '<span class="qtag tag-src-ai">AI训练师题库</span>';
}

function showQ() {
  const empty = document.getElementById('empty');
  const qnum  = document.getElementById('qnum');
  const qtext = document.getElementById('qtext');
  const optsEl= document.getElementById('opts');
  const hint  = document.getElementById('multi-hint');
  const sub   = document.getElementById('submit');
  const res   = document.getElementById('result');

  if (!queue.length) {
    empty.style.display = 'block';
    qnum.innerHTML = '';
    qtext.textContent = '';
    optsEl.innerHTML = '';
    hint.style.display = 'none';
    sub.style.display = 'none';
    res.style.display = 'none';
    return;
  }
  empty.style.display = 'none';
  sub.style.display = 'inline-block';

  const q = queue[qi];
  const typeCls = {'单选':'tag-单选','多选':'tag-多选','判断':'tag-判断'}[q.type]||'tag-单选';
  qnum.innerHTML = `<span>第 ${qi+1} / ${queue.length} 题</span>
    <span class="qtag ${typeCls}">${q.type}</span>
    ${srcTag(q.src)}
    <span style="color:#9ca3af;font-size:11px">${q.cat}</span>`;
  qtext.textContent = q.text;
  hint.style.display = q.type==='多选' ? 'block' : 'none';

  const keys = Object.keys(q.opts);
  const p = prog[q.id];
  submitted = !!p;
  selOpts = new Set(p ? p.ans : []);

  optsEl.innerHTML = keys.map(k => {
    let cls = 'opt';
    if (submitted) {
      const correct = Array.isArray(q.ans) ? q.ans : [q.ans];
      if (correct.includes(k)) cls += ' correct';
      else if (selOpts.has(k)) cls += ' wrong';
    } else if (selOpts.has(k)) {
      cls += ' sel';
    }
    return `<div class="${cls}" id="o${k}" onclick="pick('${k}')">
      <span class="opt-key">${k}</span>
      <span class="opt-txt">${q.opts[k]}</span>
    </div>`;
  }).join('');

  sub.disabled = submitted || selOpts.size === 0;
  if (submitted) showResult(q, p.ok);
  else { res.style.display = 'none'; }
  updateHeader();
}

function pick(k) {
  if (submitted) return;
  const q = queue[qi];
  if (q.type === '多选') {
    selOpts.has(k) ? selOpts.delete(k) : selOpts.add(k);
  } else {
    selOpts.clear(); selOpts.add(k);
  }
  document.querySelectorAll('.opt').forEach(el => {
    const key = el.id.replace('o','');
    el.classList.toggle('sel', selOpts.has(key));
  });
  document.getElementById('submit').disabled = selOpts.size === 0;
}

function submit() {
  if (!queue.length || submitted) return;
  const q = queue[qi];
  const correct = Array.isArray(q.ans) ? [...q.ans].sort().join('') : q.ans;
  const given   = [...selOpts].sort().join('');
  const ok = correct === given;
  prog[q.id] = { ans: [...selOpts], ok };
  try { localStorage.setItem(LS_KEY, JSON.stringify(prog)); } catch(e) {}
  submitted = true;
  document.getElementById('submit').disabled = true;
  // color options
  Object.keys(q.opts).forEach(k => {
    const el = document.getElementById('o'+k);
    if (!el) return;
    el.classList.remove('sel');
    const corArr = Array.isArray(q.ans) ? q.ans : [q.ans];
    if (corArr.includes(k)) el.classList.add('correct');
    else if (selOpts.has(k)) el.classList.add('wrong');
  });
  showResult(q, ok);
  // update dot
  const d = document.getElementById('d'+qi);
  if (d) { d.classList.remove('ans-ok','ans-ng'); d.classList.add(ok?'ans-ok':'ans-ng'); }
  updateHeader();
  buildCatList();
}

function showResult(q, ok) {
  const res = document.getElementById('result');
  const rh  = document.getElementById('rh');
  const rn  = document.getElementById('rnote');
  res.style.display = 'block';
  res.className = ok ? 'ok' : 'ng';
  const correctStr = Array.isArray(q.ans) ? q.ans.join('、') : q.ans;
  rh.textContent = ok ? '✅ 回答正确！' : `❌ 回答错误，正确答案：${correctStr}`;
  rn.textContent = q.note ? `解析：${q.note}` : '（暂无解析）';
}

function next() {
  if (qi < queue.length - 1) {
    qi++;
    selOpts.clear();
    submitted = false;
    highlightDot();
    showQ();
  }
}

// ─── header stats ────────────────────────────────────────────────
function updateHeader() {
  const all = filteredQuestions();
  const done = all.filter(q => prog[q.id] != null);
  const ok   = done.filter(q => prog[q.id].ok);
  document.getElementById('hdone').textContent = done.length;
  document.getElementById('htotal').textContent = all.length;
  const acc = done.length ? Math.round(ok.length/done.length*100) : 0;
  document.getElementById('hacc').textContent = done.length ? acc+'%' : '-';
  document.getElementById('hbarf').style.width = (done.length/Math.max(all.length,1)*100)+'%';
}

function resetAll() {
  if (!confirm('确定重置所有进度？')) return;
  prog = {};
  try { localStorage.removeItem(LS_KEY); } catch(e) {}
  buildCatList();
  buildQueue();
  updateHeader();
}

// ─── quick preview ────────────────────────────────────────────────
let pvPool = [];
const PV_PAGE = 80; // items per batch
let pvShown = 0;

function esc(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function openPreview() {
  const srcSel = document.getElementById('pvSrcFilter');
  srcSel.innerHTML = '<option value="">全部来源</option>' +
    ALLSRCS.map(s => `<option value="${esc(s)}">${esc(s)}</option>`).join('');
  const catSel = document.getElementById('pvCatFilter');
  catSel.innerHTML = '<option value="">全部分类</option>' +
    ALLCATS.map(c => `<option value="${esc(c)}">${esc(c)}</option>`).join('');
  if (curCat) catSel.value = curCat;
  if (curSrc !== '全部') srcSel.value = curSrc;
  document.getElementById('pvSearch').value = '';
  document.getElementById('pvTypeFilter').value = '';
  filterPreview();
  document.getElementById('pvOverlay').classList.add('open');
}

function closePreview() {
  document.getElementById('pvOverlay').classList.remove('open');
}

function pvBgClick(e) {
  if (e.target === document.getElementById('pvOverlay')) closePreview();
}

function filterPreview() {
  const kw   = document.getElementById('pvSearch').value.trim().toLowerCase();
  const type = document.getElementById('pvTypeFilter').value;
  const src  = document.getElementById('pvSrcFilter').value;
  const cat  = document.getElementById('pvCatFilter').value;

  pvPool = ALLQ.filter(q => {
    if (type && q.type !== type) return false;
    if (src  && q.src  !== src)  return false;
    if (cat  && q.cat  !== cat)  return false;
    if (kw) {
      const hay = (q.text + ' ' + Object.values(q.opts).join(' ') + ' ' + (q.note||'')).toLowerCase();
      if (!hay.includes(kw)) return false;
    }
    return true;
  });
  pvShown = 0;
  renderPreview(true);
}

function makePvcCard(q, idx) {
  const correct = Array.isArray(q.ans) ? q.ans : (q.ans ? q.ans.split('') : []);
  const ansStr  = correct.join('、') || '—';
  const typeCls = {'单选':'tag-单选','多选':'tag-多选','判断':'tag-判断'}[q.type] || 'tag-单选';
  const srcCls  = q.src === '泰迪云课堂' ? 'tag-src-td' : 'tag-src-ai';
  const uid     = String(q.id).replace(/[^a-zA-Z0-9_-]/g, '_');

  const wrap = document.createElement('div');
  wrap.className = 'pvc';

  // ── head ──
  const head = document.createElement('div');
  head.className = 'pvc-head';
  head.onclick = () => pvToggle(uid);

  const seq = document.createElement('span');
  seq.className = 'pvc-seq';
  seq.textContent = idx + 1;

  const mid = document.createElement('div');
  mid.style.flex = '1';

  const tagsDiv = document.createElement('div');
  tagsDiv.className = 'pvc-tags';
  const t1 = document.createElement('span'); t1.className = `qtag ${typeCls}`; t1.textContent = q.type;
  const t2 = document.createElement('span'); t2.className = `qtag ${srcCls}`;  t2.textContent = q.src;
  const t3 = document.createElement('span'); t3.style.cssText = 'font-size:11px;color:#9ca3af;padding-top:2px'; t3.textContent = q.cat;
  tagsDiv.append(t1, t2, t3);

  const qDiv = document.createElement('div');
  qDiv.className = 'pvc-q';
  qDiv.textContent = q.text;

  mid.append(tagsDiv, qDiv);

  const ansSpan = document.createElement('span');
  ansSpan.className = 'pvc-ans';
  ansSpan.textContent = '答：' + ansStr;

  const tog = document.createElement('span');
  tog.className = 'pvc-toggle';
  tog.id = 'pvt_' + uid;
  tog.textContent = '▼';

  head.append(seq, mid, ansSpan, tog);

  // ── body ──
  const body = document.createElement('div');
  body.className = 'pvc-body';
  body.id = 'pvb_' + uid;

  const optsDiv = document.createElement('div');
  optsDiv.className = 'pvc-opts';
  Object.entries(q.opts).forEach(([k, v]) => {
    const row = document.createElement('div');
    row.className = 'pvc-opt' + (correct.includes(k) ? ' right' : '');
    const kk = document.createElement('span'); kk.className = 'pvc-opt-k'; kk.textContent = k + '.';
    const vv = document.createElement('span'); vv.textContent = v;
    row.append(kk, vv);
    optsDiv.appendChild(row);
  });
  body.appendChild(optsDiv);

  if (q.note) {
    const noteWrap = document.createElement('div');
    noteWrap.className = 'pvc-note';
    const noteTitle = document.createElement('div');
    noteTitle.className = 'pvc-notitle';
    noteTitle.textContent = '💡 解析';
    const noteTxt = document.createElement('div');
    noteTxt.textContent = q.note;
    noteWrap.append(noteTitle, noteTxt);
    body.appendChild(noteWrap);
  }

  wrap.append(head, body);
  return wrap;
}

function renderPreview(reset) {
  const list  = document.getElementById('pvList');
  const empty = document.getElementById('pvEmpty');
  const stats = document.getElementById('pvStats');
  const total = pvPool.length;
  const withNote = pvPool.filter(q => q.note).length;
  stats.innerHTML = `共 <b>${total}</b> 题 &nbsp;|&nbsp; 有解析 <b>${withNote}</b> 题 &nbsp;|&nbsp; 点击题目展开选项与解析`;

  if (reset) {
    [...list.children].forEach(c => { if (c.id !== 'pvEmpty' && c.id !== 'pvMore') c.remove(); });
    const old = document.getElementById('pvMore');
    if (old) old.remove();
  }

  if (!total) { empty.style.display = 'block'; return; }
  empty.style.display = 'none';

  const end = Math.min(pvShown + PV_PAGE, total);
  const frag = document.createDocumentFragment();
  for (let i = pvShown; i < end; i++) {
    frag.appendChild(makePvcCard(pvPool[i], i));
  }
  pvShown = end;

  // remove old "load more" btn before re-inserting
  const oldMore = document.getElementById('pvMore');
  if (oldMore) oldMore.remove();

  list.insertBefore(frag, empty);

  if (pvShown < total) {
    const btn = document.createElement('button');
    btn.id = 'pvMore';
    btn.textContent = `加载更多（还有 ${total - pvShown} 题）`;
    btn.style.cssText = 'width:100%;padding:10px;margin:8px 0;border:1.5px dashed #a78bfa;background:#faf5ff;color:#6d28d9;border-radius:8px;font-size:13px;cursor:pointer;font-weight:600';
    btn.onclick = () => renderPreview(false);
    list.insertBefore(btn, empty);
  }
}

function pvToggle(uid) {
  const body   = document.getElementById('pvb_' + uid);
  const toggle = document.getElementById('pvt_' + uid);
  if (!body) return;
  const isOpen = body.classList.contains('open');
  body.classList.toggle('open', !isOpen);
  toggle.classList.toggle('open', !isOpen);
}
